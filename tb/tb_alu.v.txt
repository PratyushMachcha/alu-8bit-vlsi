module tb_alu;

reg [7:0] a, b;
reg [2:0] sel;
wire [7:0] y;

alu_8bit uut (.a(a), .b(b), .sel(sel), .y(y));

initial begin
    $dumpfile("dump.vcd");
    $dumpvars(0, tb_alu);
    $monitor("Time=%0t sel=%b a=%d b=%d y=%d", $time, sel, a, b, y);

    a = 8'd10; b = 8'd5;

    sel = 3'b000; #10;
    sel = 3'b001; #10;
    sel = 3'b010; #10;
    sel = 3'b011; #10;
    sel = 3'b100; #10;
    sel = 3'b101; #10;
    sel = 3'b110; #10;
    sel = 3'b111; #10;

    #10 $finish;
end

endmodule
